(function () {
  "use strict";

  var video    = document.getElementById("video");
  var canvas   = document.getElementById("out");
  var ctx      = canvas.getContext("2d", { willReadFrequently: true, alpha: false });
  var stage    = document.getElementById("stage");
  var empty    = document.getElementById("empty");
  var playBtn  = document.getElementById("play");
  var muteBtn  = document.getElementById("mute");
  var icoOn    = document.getElementById("icoVolOn");
  var icoOff   = document.getElementById("icoVolOff");
  var fileIn   = document.getElementById("file");
  var saveBtn  = document.getElementById("save");
  var saveTxt  = document.getElementById("saveTxt");
  var statusEl = document.getElementById("status");

  var loaded = false;
  var needsFrame = true;
  var vigCache = {};
  var lastUrl = null;

  /* ---------- tone curve: deep blacks, soft highlights ---------- */
  var K = 0.55;
  var curve = new Uint8ClampedArray(256);
  for (var i = 0; i < 256; i++) {
    var t = i / 255;
    var s = t + K * t * (1 - t) * (2 * t - 1); // filmic S-curve
    curve[i] = s * 0.955 * 255;                // deepen overall
  }
  /* cool grade baked per channel: -2 red / 0 green / +4 blue */
  function ci(v){ return v < 0 ? 0 : v > 255 ? 255 : v; }
  var lutR = new Uint8ClampedArray(256);
  var lutG = new Uint8ClampedArray(256);
  var lutB = new Uint8ClampedArray(256);
  for (var j = 0; j < 256; j++) {
    lutR[j] = curve[ci(j - 2)];
    lutG[j] = curve[ci(j)];
    lutB[j] = curve[ci(j + 4)];
  }

  /* ---------- film grain tile ---------- */
  var T = 256;
  var grain = new Float32Array(T * T);
  for (var g = 0; g < grain.length; g++) {
    grain[g] = (Math.random() * 2 - 1) * 9;
  }

  function buildVig(w, h) {
    var arr = new Float32Array(w * h);
    var cx = w / 2, cy = h / 2;
    var maxD = Math.sqrt(cx * cx + cy * cy) || 1;
    for (var y = 0; y < h; y++) {
      for (var x = 0; x < w; x++) {
        var dx = (x + 0.5 - cx) / maxD;
        var dy = (y + 0.5 - cy) / maxD;
        var d = Math.sqrt(dx * dx + dy * dy);
        var t = (d - 0.55) / (1.02 - 0.55);
        t = t < 0 ? 0 : t > 1 ? 1 : t;
        t = t * t * (3 - 2 * t);
        arr[y * w + x] = 1 - t * 0.36;
      }
    }
    return arr;
  }
  function getVig(w, h) {
    var k = w + "x" + h;
    if (!vigCache[k]) vigCache[k] = buildVig(w, h);
    return vigCache[k];
  }

  function fit() {
    var r = stage.getBoundingClientRect();
    var vw = video.videoWidth || 16;
    var vh = video.videoHeight || 9;
    var s = Math.min(r.width / vw, r.height / vh);
    if (!isFinite(s) || s <= 0) return;
    var w = Math.max(2, Math.round(vw * s));
    var h = Math.max(2, Math.round(vh * s));
    var cap = 1280;
    if (w > cap) { h = Math.round(h * cap / w); w = cap; }
    canvas.width = w;
    canvas.height = h;
    canvas.style.width = w + "px";
    canvas.style.height = h + "px";
    getVig(w, h);
    needsFrame = true;
  }

  function paint(c2d, w, h, vigArr) {
    c2d.drawImage(video, 0, 0, w, h);
    var img = c2d.getImageData(0, 0, w, h);
    var data = img.data;
    var ox = (Math.random() * T) | 0;
    var oy = (Math.random() * T) | 0;
    var KS = 0.6; /* saturation kept: 60% -> muted cool-gray */

    for (var y = 0; y < h; y++) {
      var row = y * w;
      var gRow = ((y + oy) & 255) << 8;
      for (var x = 0; x < w; x++) {
        var pi = row + x;
        var o = pi * 4;
        var r = data[o], gg = data[o + 1], b = data[o + 2];

        var L = r * 0.2126 + gg * 0.7152 + b * 0.0722;
        r  = L + (r - L) * KS;
        gg = L + (gg - L) * KS;
        b  = L + (b - L) * KS;

        var v = vigArr[pi];
        var n = grain[gRow | ((x + ox) & 255)];

        data[o]     = lutR[r | 0] * v + n;
        data[o + 1] = lutG[gg | 0] * v + n;
        data[o + 2] = lutB[b | 0] * v + n;
      }
    }
    c2d.putImageData(img, 0, 0);
  }

  function draw() {
    var w = canvas.width, h = canvas.height;
    if (w < 2 || h < 2) return;
    paint(ctx, w, h, getVig(w, h));
  }

  function loop() {
    requestAnimationFrame(loop);
    if (!loaded || video.readyState < 2) return;
    if (!video.paused) {
      draw();
    } else if (needsFrame) {
      needsFrame = false;
      draw();
    }
  }

  function updateOverlay() {
    playBtn.classList.toggle("show", loaded && video.paused);
  }
  function updateMute() {
    icoOn.style.display  = video.muted ? "none" : "block";
    icoOff.style.display = video.muted ? "block" : "none";
  }
  function togglePlay() {
    if (video.paused) { video.play(); } else { video.pause(); }
  }

    var f = fileIn.files && fileIn.files[0];

    if (!f) return;
    if (lastUrl) { try { URL.revokeObjectURL(lastUrl); } catch (e) {} lastUrl = null; }
    resetExport();
    saveBtn.disabled = true;
    saveBtn.classList.remove("show");
    saveTxt.textContent = "\u2B07\uFE0F Save Video to Phone";
    statusEl.textContent = "";
    if (window.URL && URL.createObjectURL) {
      var url = URL.createObjectURL(f);
      video.src = url;
    } else if (video.canPlayType) {
      var rf = new FileReader();
      rf.onload = function () { video.src = rf.result; };
      rf.readAsDataURL(f);
    }
    video.load();
  });

  video.addEventListener("loadedmetadata", function () { fit(); });
  video.addEventListener("loadeddata", function () {
    loaded = true;
    empty.style.display = "none";
    muteBtn.classList.add("show");
    saveBtn.classList.add("show");
    saveBtn.disabled = false;
    fit();
    draw();
    updateMute();
    /* first frame shows immediately; try muted autoplay for instant motion */
    video.muted = true;
    updateMute();
    var p;
    try { p = video.play(); } catch (e) { p = null; }
    if (p && p.then) {
      p.then(updateOverlay).catch(function () {
        video.muted = false;
        updateMute();
        updateOverlay();
      });
    } else {
      video.muted = false;
      updateMute();
      updateOverlay();
    }
  });
  video.addEventListener("play",  updateOverlay);
  video.addEventListener("pause", function () { needsFrame = true; updateOverlay(); });
  video.addEventListener("ended", updateOverlay);
  video.addEventListener("seeked", function () { needsFrame = true; });
  video.addEventListener("volumechange", updateMute);

  playBtn.addEventListener("click", function (e) { e.stopPropagation(); togglePlay(); });
  muteBtn.addEventListener("click", function (e) {
    e.stopPropagation();
    video.muted = !video.muted;
  });
  stage.addEventListener("click", function () { if (loaded) togglePlay(); });

  window.addEventListener("resize", function () { fit(); });
  window.addEventListener("orientationchange", function () { setTimeout(fit, 120); });
  if (window.visualViewport) {
    window.visualViewport.addEventListener("resize", function () { fit(); });
  }

  /* ---------- EXPORT: iOS-safe filtered video export ---------- */
  function isIOS() {
    return /iPad|iPhone|iPod/.test(navigator.userAgent) ||
      (navigator.platform === "MacIntel" && navigator.maxTouchPoints > 1);
  }

  function pickMime() {
    if (!window.MediaRecorder) return null;
    var types = isIOS()
      ? ["video/mp4;codecs=avc1,mp4a.40.2", "video/mp4;codecs=avc1", "video/mp4"]
      : ["video/webm;codecs=vp9", "video/webm;codecs=vp8", "video/webm", "video/mp4"];
    for (var i = 0; i < types.length; i++) {
      try {
        if (MediaRecorder.isTypeSupported(types[i])) return types[i];
      } catch (e) {}
    }
    return null;
  }

  function setStatus(t){ statusEl.textContent = t; }

  function shareOrDownload(blob, name) {
    var fl = null;
    try { fl = new File([blob], name, { type: blob.type || "video/mp4" }); } catch (e) {}

    /* iPhone: Share Sheet is the most reliable way to put the finished
       video into Photos/Files. */
    if (isIOS() && navigator.share && fl) {
      try {
        if (!navigator.canShare || navigator.canShare({ files: [fl] })) {
          navigator.share({ files: [fl], title: "NOIR filtered video" })
            .then(function () {
              setStatus("Done — choose Save Video in the Share menu");
            })
            .catch(function () {});
          return;
        }
      } catch (e) {}
    }

    var url = null;
    try { url = URL.createObjectURL(blob); } catch (e) {}
    if (url) {
      var a = document.createElement("a");
      a.href = url;
      a.download = name;
      a.rel = "noopener";
      document.body.appendChild(a);
      a.click();
      setTimeout(function () {
        try { document.body.removeChild(a); } catch (e) {}
        try { URL.revokeObjectURL(url); } catch (e) {}
      }, 1000);
    }
  }

  var exportState = "idle";
  var exportBlob = null;
  var exportName = null;

  function resetExport() {
    exportState = "idle";
    exportBlob = null;
    exportName = null;
  }

  function doExport() {
    if (exportState !== "idle" || !loaded) return;

    var mime = pickMime();
    if (!mime) {
      setStatus("This browser cannot export video");
      return;
    }

    /* captureStream/requestFrame are not equally reliable on iOS.
       Stop using a 0fps canvas stream there; capture a normal frame stream
       and explicitly request frames when available. */
    if (!canvas.captureStream) {
      setStatus("Video export is not supported on this iPhone browser");
      return;
    }

    exportState = "exporting";
    saveBtn.disabled = true;
    saveTxt.textContent = "\u2B07\uFE0F Processing video\u2026";
    setStatus("Processing video\u2026 0%");

    try { video.pause(); } catch (e) {}

    var w = video.videoWidth, h = video.videoHeight;
    if (!w || !h) {
      exportState = "idle";
      saveBtn.disabled = false;
      setStatus("Export failed");
      return;
    }

    var cap = isIOS() ? 960 : 1280;
    if (w > cap) { h = Math.round(h * cap / w); w = cap; }

    var off = document.createElement("canvas");
    off.width = w;
    off.height = h;
    var octx = off.getContext("2d", { alpha: false, willReadFrequently: true });
    if (!octx) {
      exportState = "idle";
      saveBtn.disabled = false;
      setStatus("Export failed");
      return;
    }

    var vigArr = getVig(w, h);
    var stream;
    try {
      stream = off.captureStream(30);
    } catch (e) {
      exportState = "idle";
      saveBtn.disabled = false;
      setStatus("Could not start video export");
      return;
    }

    var rec;
    try {
      rec = new MediaRecorder(stream, {
        mimeType: mime,
        videoBitsPerSecond: isIOS() ? 2500000 : Math.max(1500000, Math.round(w * h * 0.14))
      });
    } catch (e) {
      try { rec = new MediaRecorder(stream); }
      catch (e2) {
        exportState = "idle";
        saveBtn.disabled = false;
        setStatus("Could not start video export");
        return;
      }
    }

    var chunks = [];
    rec.ondataavailable = function (e) {
      if (e.data && e.data.size > 0) chunks.push(e.data);
    };

    var raf = 0;
    var done = false;
    var duration = video.duration || 0;
    var startedAt = Date.now();

    function cleanup() {
      cancelAnimationFrame(raf);
      try { video.pause(); } catch (e) {}
      video.removeEventListener("ended", finish);
      try {
        stream.getTracks().forEach(function(t) { t.stop(); });
      } catch (e) {}
    }

    function finish() {
      if (done) return;
      done = true;
      cleanup();
      try { rec.stop(); } catch (e) {}
    }

    rec.onstop = function () {
      var mimePlain = (rec.mimeType || mime || "video/mp4").split(";")[0];
      var ext = /mp4/i.test(mimePlain) ? "mp4" : "webm";
      var blob = new Blob(chunks, { type: mimePlain });

      if (blob.size < 4096 || !chunks.length) {
        exportState = "idle";
        saveBtn.disabled = false;
        saveTxt.textContent = "\u2B07\uFE0F Save Video to Phone";
        setStatus("Export produced no data");
        return;
      }

      exportBlob = blob;
      exportName = "noir-filtered-" +
        new Date().toISOString().slice(0, 19).replace(/[:T]/g, "-") + "." + ext;
      exportState = "ready";
      saveBtn.disabled = false;
      saveTxt.textContent = "\u2B07\uFE0F Save Filtered Video";
      setStatus(isIOS()
        ? "Ready — tap Save, then choose Save Video"
        : "Ready — tap Save to get the filtered video");
    };

    function paintFrame() {
      if (exportState !== "exporting") return;
      paint(octx, w, h, vigArr);
      try {
        var track = stream.getVideoTracks()[0];
        if (track && track.requestFrame) track.requestFrame();
      } catch (e) {}
    }

    function tick() {
      if (exportState !== "exporting") return;

      paintFrame();

      var cur = video.currentTime || 0;
      var pct = duration > 0
        ? Math.min(99, Math.round((cur / duration) * 100))
        : 0;
      setStatus("Processing video\u2026 " + pct + "%");

      /* Some iOS versions can fail to deliver "ended". */
      if ((duration > 0 && cur >= duration - 0.08) ||
          Date.now() - startedAt > Math.max(15000, (duration + 5) * 1000)) {
        finish();
        return;
      }
      raf = requestAnimationFrame(tick);
    }

    video.addEventListener("ended", finish);
    paintFrame();

    try {
      rec.start(250);
    } catch (e) {
      exportState = "idle";
      saveBtn.disabled = false;
      setStatus("Could not start recording");
      return;
    }

    video.currentTime = 0;

    /* Do not require audio capture on iOS. The old implementation depended
       on video.captureStream() audio tracks, which is a common iOS failure
       point. The filtered video remains exportable even without audio. */
    try { video.muted = true; } catch (e) {}
    updateMute();

    var playPromise = null;
    try { playPromise = video.play(); } catch (e) {}

    if (playPromise && playPromise.catch) {
      playPromise.catch(function () {
        finish();
        exportState = "idle";
        saveBtn.disabled = false;
        setStatus("Tap Play first, then try Save again");
      });
    }

    raf = requestAnimationFrame(tick);
  }

  saveBtn.addEventListener("click", function () {
    if (exportState === "ready" && exportBlob) {
      shareOrDownload(exportBlob, exportName);
      setStatus("Saved \u2014 check your downloads/files");
    } else if (exportStat
